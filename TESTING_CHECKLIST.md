## 18. Как запускать (шпаргалка)

# статика/типы
npm run lint            # tsc --noEmit
npm run build           # vite build + esbuild server.ts → dist/
pm2 restart learning-platform
curl -s localhost:3000/api/health          # {"ok":true}

# API-смоук (замени на свой токен)
T=$(curl -s -X POST localhost:3000/api/auth/login \
  -H 'Content-Type: application/json' \
  -d '{"username":"admin","password":"<pass>"}' | python3 -c 'import sys,json;print(json.load(sys.stdin)["access_token"])')
curl -s localhost:3000/api/admin/users -H "Authorization: Bearer $T"

# проверка неопределённых имён в .jsx
npx tsc -p /tmp/opencode/tsconfig.check.json   # grep TS2304

# E2E: headless Chrome + CDP-харнесс (/tmp/opencode/ui.js, lesson.js)
#   - токены ролей в /tmp/opencode/roles.json
#   - кейсы: {role, name, url, actions:[{js,wait}], eval}
#   - ловушки: \n и \d в eval-строках пишутся как \\n/\\d;
#     модалка = '.modal' / '.modal-back'; приветствие закрывается кнопкой «Понятно»
